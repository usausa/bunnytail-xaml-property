pipeline {
  agent any

  parameters {
    booleanParam(name: 'CLEAN', defaultValue: false)
  }

  options {
    ansiColor('xterm')
    timestamps()

    timeout(time: 5, unit: 'MINUTES')
    buildDiscarder(logRotator(numToKeepStr: '50', artifactNumToKeepStr: '50'))
  }

  triggers {
    pollSCM('5,15,25,35,45,55 * * * *')
  }

  stages {
    stage('Checkout') {
      steps {
        script {
          if (params.CLEAN) {
            cleanWs()
          }
        }

        checkout([
          $class: 'GitSCM',
          branches: [[name: '**']],
          userRemoteConfigs: [[url: 'https://github.com/usausa/bunnytail-xaml-property']],
          extensions: [
            [$class: 'DisableRemotePoll']]])
      }
    }

    stage('Build') {
      steps {
        bat '''
chcp 65001
rmdir /s /q R:\\Temp\\Generator_XamlProperty
mkdir R:\\Temp\\Generator_XamlProperty
SET TEMP=R:\\Temp\\Generator_XamlProperty
SET TMP=R:\\Temp\\Generator_XamlProperty

dotnet clean BunnyTail.XamlProperty.slnx -c Release
dotnet build BunnyTail.XamlProperty.slnx -c Release -nodeReuse:false

rmdir /s /q Publish
dotnet pack --no-restore BunnyTail.XamlProperty.Wpf.Generator\\BunnyTail.XamlProperty.Wpf.Generator.csproj -o Publish -c Release || exit /b 1
dotnet pack --no-restore BunnyTail.XamlProperty.Maui.Generator\\BunnyTail.XamlProperty.Maui.Generator.csproj -o Publish -c Release || exit /b 1
dotnet pack --no-restore BunnyTail.XamlProperty.Avalonia.Generator\\BunnyTail.XamlProperty.Avalonia.Generator.csproj -o Publish -c Release || exit /b 1
'''
      }
    }

    stage('Inspect') {
      steps {
        bat '''
chcp 65001
SET TEMP=R:\\Temp\\Generator_XamlProperty
SET TMP=R:\\Temp\\Generator_XamlProperty

jb inspectcode BunnyTail.XamlProperty.slnx -f=xml -o=results.xml --no-build --no-swea --properties:Configuration=Release
'''
      }
    }

    stage('Test') {
      steps {
        catchError(buildResult: 'UNSTABLE', stageResult: 'UNSTABLE') {
          bat '''
chcp 65001
SET TEMP=R:\\Temp\\Generator_XamlProperty
SET TMP=R:\\Temp\\Generator_XamlProperty

rmdir /s /q BunnyTail.XamlProperty.Wpf.Tests\\bin\\Release\\net10.0-windows\\TestResults
rmdir /s /q BunnyTail.XamlProperty.Maui.Tests\\bin\\Release\\net10.0\\TestResults
rmdir /s /q BunnyTail.XamlProperty.Avalonia.Tests\\bin\\Release\\net10.0\\TestResults

dotnet run --project BunnyTail.XamlProperty.Wpf.Tests -c Release --report-xunit-trx --report-xunit-trx-filename BunnyTail.XamlProperty.Wpf.Tests.trx --coverage --coverage-settings CodeCoverage.runsettings --coverage-output coverage.cobertura.xml || exit /b 1
dotnet run --project BunnyTail.XamlProperty.Maui.Tests -c Release --report-xunit-trx --report-xunit-trx-filename BunnyTail.XamlProperty.Maui.Tests.trx --coverage --coverage-settings CodeCoverage.runsettings --coverage-output coverage.cobertura.xml || exit /b 1
dotnet run --project BunnyTail.XamlProperty.Avalonia.Tests -c Release --report-xunit-trx --report-xunit-trx-filename BunnyTail.XamlProperty.Avalonia.Tests.trx --coverage --coverage-settings CodeCoverage.runsettings --coverage-output coverage.cobertura.xml || exit /b 1
'''
        }
      }
    }

    stage('Report') {
      steps {
        // Cobertura coverage report
        recordCoverage(
          tools: [[parser: 'COBERTURA', pattern: '**/*.cobertura.xml']],
          id: 'cobertura',
          name: 'Coverage')

        // Publish MSTest test result report
        step([
          $class: 'MSTestPublisher',
          testResultsFile:"**/*.trx",
          failOnError: true,
          keepLongStdio: false])

        // Record compiler warnings and static analysis results
        recordIssues(
          tools: [
            msBuild(reportEncoding: 'UTF-8'),
            resharperInspectCode(pattern: 'results.xml', reportEncoding: 'UTF-8')],
          sourceCodeEncoding: 'UTF-8',
          trendChartType: 'TOOLS_ONLY',
          qualityGates: [[threshold: 1, type: 'TOTAL', unstable: true]])
        recordIssues(
          tools: [taskScanner(includePattern: '**/*.cs', lowTags: 'TODO')],
          sourceCodeEncoding: 'UTF-8',
          trendChartType: 'TOOLS_ONLY')

        // Step counter
        bat(script: 'cloc.exe . --xml --out=cloc.xml --by-file --include-lang=C#,Razor,SQL,JavaScript,CSS --vcs=git --exclude-dir=lib --exclude-ext=min.js,min.css --exclude-content="auto-generated"')
        sloccountPublish(
          encoding: '',
         ignoreBuildFailure: true,
         pattern: 'cloc.xml')
      }
    }
  }

  post {
    success {
      // Send build artifacts to a windows share
      script {
        if (isReleaseBranch(getCurrentBranch())) {
          cifsPublisher(
            publishers: [[
              configName: 'Artifact',
              transfers: [[
                  sourceFiles: 'Publish/**/*',
                  removePrefix: 'Publish/',
                  remoteDirectory: 'NuGetPackages',
                  excludes: '',
                  patternSeparator: '[, ]+',
                  cleanRemote: false]]]])
        }
      }

      // Signal
      postSignal('green')
      // Slack notifications
      postSlackSend('good', 'Success')
    }

    unstable {
      // Signal
      postSignal('yellow')
      // Slack notifications
      postSlackSend('warning', 'Unstable')
    }

    failure {
      // Signal
      postSignal('red')
      // Slack notifications
      postSlackSend('danger', 'Failure')
    }
  }
}

def getCurrentBranch() {
  def lines = bat(
    script: 'git name-rev --name-only HEAD',
    returnStdout: true
  ).split('\n')
  lines[lines.size() - 1]
}

def isReleaseBranch(branch) {
  branch.contains('master') || branch.contains('main') || ('tags')
}

def postSlackSend(color, result) {
  slackSend(color: color, message: "${env.JOB_NAME} #${env.BUILD_NUMBER} ${result} [${currentBuild.durationString}] (<${env.BUILD_URL}|Open>)")
}

def postSignal(color) {
  withEnv(['JENKINS_NODE_COOKIE=do_not_kill']) {
    bat """
start \"patlite\" patlite_${color}.bat
"""
  }
  //httpRequest(url: "http://10.13.8.13:8080/play/${color}.dat", httpMode: 'POST')
}
